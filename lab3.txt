-- Part A: Database and Table Setup
-- 1. Create database and tables
CREATE TABLE IF NOT EXISTS employees (
    emp_id SERIAL PRIMARY KEY,
    first_name VARCHAR(50) NOT NULL,
    last_name VARCHAR(50) NOT NULL,
    department VARCHAR(50) DEFAULT 'Unassigned',
    salary INT DEFAULT 40000,
    hire_date DATE DEFAULT CURRENT_DATE,
    status VARCHAR(20) DEFAULT 'Active'
);

CREATE TABLE IF NOT EXISTS departments (
    dept_id SERIAL PRIMARY KEY,
    dept_name VARCHAR(50) UNIQUE NOT NULL,
    budget INT DEFAULT 0,
    manager_id INT
);

CREATE TABLE IF NOT EXISTS projects (
    project_id SERIAL PRIMARY KEY,
    project_name VARCHAR(100) NOT NULL,
    dept_id INT,
    start_date DATE,
    end_date DATE,
    budget INT
);

-- 7. UPDATE with arithmetic expressions[cite: 4]

-- 8. UPDATE with WHERE clause and multiple conditions[cite: 4]

-- 9. UPDATE using CASE expression[cite: 4]
UPDATE employees 
SET department = CASE 
    WHEN salary > 80000 THEN 'Management'
    WHEN salary BETWEEN 50000 AND 80000 THEN 'Senior'
    ELSE 'Junior'
END;

-- 10. UPDATE with DEFAULT[cite: 4]

-- 11. UPDATE with subquery[cite: 4]
UPDATE departments d
SET budget = CAST((SELECT AVG(e.salary) FROM employees e WHERE e.department = d.dept_name) * 1.20 AS INT)
WHERE EXISTS (SELECT 1 FROM employees e WHERE e.department = d.dept_name);

-- 12. UPDATE multiple columns[cite: 4]

-- Part G: Advanced DML Patterns


-- 23. Conditional INSERT[cite: 4]
INSERT INTO employees (first_name, last_name, department, salary)
SELECT 'Bruce', 'Wayne', 'Management', 100000
WHERE NOT EXISTS (
    SELECT 1 FROM employees WHERE first_name = 'Bruce' AND last_name = 'Wayne'
);

-- 24. UPDATE with JOIN logic using subqueries[cite: 4]
UPDATE employees 
SET salary = CAST(salary * (
    CASE 
        WHEN (SELECT budget FROM departments WHERE dept_name = employees.department) > 100000 THEN 1.10
        ELSE 1.05
    END
) AS INT)
WHERE department IN (SELECT dept_name FROM departments);

-- 25. Bulk operations[cite: 4]
WITH inserted_users AS (
    INSERT INTO employees (first_name, last_name, department, salary) VALUES

-- 26. Data migration simulation[cite: 4]
CREATE TABLE IF NOT EXISTS employee_archive (LIKE employees INCLUDING ALL);

WITH moved_rows AS (
    DELETE FROM employees 
    WHERE status = 'Inactive' 
    RETURNING *
)
INSERT INTO employee_archive 
SELECT * FROM moved_rows;

-- 27. Complex business logic[cite: 4]
UPDATE projects 
SET end_date = end_date + INTERVAL '30 days'
WHERE budget > 50000 AND dept_id IN (
    SELECT d.dept_id 
    FROM departments d 
    JOIN employees e ON d.dept_name = e.department 
    GROUP BY d.dept_id 
    HAVING COUNT(e.emp_id) > 3
);
